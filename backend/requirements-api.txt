# API service dependencies.
# The api ingests documents in-process (upload -> parse -> chunk -> embed ->
# index), so for maximal LOCAL/air-gap capability it carries the full stack:
# real local embeddings (sentence-transformers), OCR (pytesseract + PyMuPDF),
# and the document parsers/exporters. The system tesseract + LibreOffice
# binaries are installed in the Dockerfile; the MiniLM weights are baked at
# build time so the image is fully offline.

# Core API
fastapi==0.115.2
uvicorn[standard]==0.30.6
pydantic==2.9.2
python-multipart==0.0.12

# Document parsers/exporters + vector-layout geometry (all light)
python-docx==1.1.2
python-pptx==1.0.2
pypdf==5.0.1
reportlab==4.2.2
Pillow>=10.4.0
pdfplumber==0.11.10

# AWS SDK for the optional Bedrock scenario-generation path. Lightweight and
# used only when credentials resolve + a model is selected; the api still runs
# fully offline (deterministic generator) when it is unused.
boto3==1.43.104

# OCR for scanned/flattened PDFs and images. The pure-Python wheels need the
# system `tesseract` binary (installed in the Dockerfile). PyMuPDF rasterizes
# PDF pages with no system poppler dependency. OCR stays gated by OCR_ENABLED
# and ocr_available(), so a host without tesseract still degrades gracefully.
pytesseract==0.3.13
PyMuPDF==1.24.10

# Local embedding model (Apache-2.0, CPU, fully offline once the weights are
# baked into the image). Gives real semantic search instead of the hashing
# fallback. Swappable behind the EmbeddingProvider seam; never load-bearing.
sentence-transformers>=3.0,<7.0
